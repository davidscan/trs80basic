20 DEF FNA(X)=X*X+C
30 C=1:X=99
40 PRINT "CLOSURE";FNA(3);X
50 DEF FN AB(P,Q)=P*10+Q
60 PRINT "TWOARG";FNAB(3,4);FN AB(1,2)
70 DEFFNH$(A$)=A$+"!"
80 PRINT "STRFN ";FNH$("HI")
90 DEF FNR=C+1
100 PRINT "NOARG";FNR
110 DEF FNA(X)=X+100
120 PRINT "REDEF";FNA(3)
130 DEF FNF(X)=FNA(X)*2
140 PRINT "NESTED";FNF(1)
150 FNZZ=77
160 PRINT "FNVAR";FNZZ;FNQ8(2)
170 ON ERROR GOTO 190
180 E1=0:DEF FNQ(X)=FNQ(X+1):Y=FNQ(1)
185 GOTO 200
190 E1=ERR/2+1:RESUME 200
200 PRINT "RECUR";E1
210 ON ERROR GOTO 230
220 E2=0:Y=FNA("S")
225 GOTO 240
230 E2=ERR/2+1:RESUME 240
240 PRINT "ARGTM";E2
250 ON ERROR GOTO 270
260 E3=0:Y$=FNH$(5)
265 GOTO 280
270 E3=ERR/2+1:RESUME 280
280 PRINT "STRTM";E3
290 END
RUN

PRINT FNA(1)
