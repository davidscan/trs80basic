REM t21: MID$ statement, OUT no-op, LPRINT/LLIST (discard mode in sweep)
10 A$="HELLO WORLD"
20 MID$(A$,7)="THERE"
30 PRINT A$
40 MID$(A$,1,3)="XYZZY"
50 PRINT A$
60 DIM B$(3): B$(2)="ABCDEF"
70 MID$(B$(2),3,2)="12345"
80 PRINT B$(2)
90 DEFSTR C
100 C="ABCD": MID$(C,2)="XY": PRINT C
110 MID$(A$,10)="LONGTAIL"
120 PRINT A$; LEN(A$)
130 OUT 255,1
140 OUT 128+127,0
150 PRINT "OUT OK"
160 LPRINT "P1";TAB(10);"COL10"
170 LPRINT "A","B"
180 LPRINT USING "##.#"; 3.14
190 PRINT "LPRINT OK"
200 ON ERROR GOTO 300
210 MID$(A$,99)="X"
220 PRINT "NOT REACHED": END
300 PRINT "FC =";(ERR=8): RESUME 310
310 END
RUN
LLIST
BYE
