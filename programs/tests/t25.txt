10 REM T25 AUTHENTIC ROM RND -- 24-BIT LCG, SEED AT 16554-16556
20 POKE 16554,0:POKE 16555,0:POKE 16556,0
30 R1=RND(0)
40 PRINT "STEP1";PEEK(16554);PEEK(16555);PEEK(16556)
50 PRINT "VAL1";R1
60 R2=RND(0):PRINT "VAL2";R2
70 REM RND(1) ALWAYS 1 (THE DIALECT TRAP)
80 T=0:FOR I=1 TO 50:T=T+RND(1):NEXT:PRINT "RNDONE";T
90 REM RND(N) STAYS IN 1..N
100 M1=99:M2=0:FOR I=1 TO 200:X=RND(6):IF X<M1 THEN M1=X
110 IF X>M2 THEN M2=X
115 NEXT
120 PRINT "RANGE";M1;M2
130 REM SEED E20F02H YIELDS MANTISSA FFFFFF -- PRINTS AS 1 (ROM QUIRK)
140 POKE 16554,2:POKE 16555,15:POKE 16556,226
150 PRINT "QUIRK";RND(0)
160 PRINT "QUIRKSEED";PEEK(16554);PEEK(16555);PEEK(16556)
170 REM RANDOM REWRITES ONLY THE MIDDLE BYTE
180 POKE 16554,7:POKE 16555,8:POKE 16556,9:RANDOM
190 PRINT "MIDONLY";PEEK(16554);PEEK(16556)
200 ON ERROR GOTO 220
210 E9=0:X=RND(-1)
215 GOTO 230
220 E9=ERR/2+1:RESUME 230
230 PRINT "NEGFC";E9
240 END
